cat /var/www/index.shtml  -> zobaczyc jakich zmiennych uzywa tabelka

pobranie
cd /root
U=https://raw.githubusercontent.com/macius714/macius714.github.io/main/cw7/apache
curl -O $U/asu1.conf
curl -O $U/asu2.conf
curl -O $U/asu3.conf

etap 1
mkdir /var/www/asu1 /var/www/asu2 /var/www/asu3
cp /var/www/index.shtml /var/www/asu1/
cp /var/www/index.shtml /var/www/asu2/
cp /var/www/index.shtml /var/www/asu3/
htpasswd -cb /var/www/htpasswd ada ada
htpasswd -b /var/www/htpasswd igor igor
cp asu1.conf asu2.conf asu3.conf /etc/apache2/sites-available/
a2enmod include ldap authnz_ldap
a2ensite asu1 asu2 asu3
apache2ctl configtest
service apache2 restart
