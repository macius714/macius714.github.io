jak nie -> w pinerc-ewa i pinerc-jan zamienic host1.asu.ia.pw.edu.pl na ip host1

pobranie
cd /root
U=https://raw.githubusercontent.com/macius714/macius714.github.io/main/cw5/host1
curl -O $U/local.conf
curl -O $U/aliases
curl -O $U/pinerc-ewa
curl -O $U/pinerc-jan
apt-get install postfix dovecot-imapd alpine telnet
  postfix: Internet Site, nazwa: asu.ia.pw.edu.pl

etap 1
postconf -e "myhostname = host1.asu.ia.pw.edu.pl"
postconf -e "mydomain = asu.ia.pw.edu.pl"
postconf -e "myorigin = asu.ia.pw.edu.pl"
postconf -e "mydestination = host1.asu.ia.pw.edu.pl, asu.ia.pw.edu.pl, localhost"
postconf -e "mynetworks = 127.0.0.0/8 192.168.0.0/16"
postconf -e "inet_interfaces = all"
service postfix restart
postconf -n
cp local.conf /etc/dovecot/
service dovecot restart
echo "Subject: test1" | sendmail ewa@asu.ia.pw.edu.pl

  a login ewa ewa
  b select inbox
  c logout
